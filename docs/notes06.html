def count_letters(string: str) -> dict[str, int]:
    ''' Returns a dictionary containing counts of each letter in string

    >>> count_letters('aaa bb c')
    {'a': 3, 'b': 2, 'c': 1}
    '''
    counts = {}                       # Discuss: counting pattern

    for letter in string:
        if not letter.isspace():      # Discuss: isspace
            if letter in counts:
                counts[letter] += 1
            else:
                counts[letter]  = 0   # Discuss: get method
            # counts[letter] = counts.get(letter, 0) + 1

    return counts

def main(arguments=sys.argv[1:], stream=sys.stdin) -> None:
    # Parse command-line options
    ignorecase = False

    # Discuss: popping from queue
    while arguments and arguments[0].startswith('-'):
        match argument := arguments.pop(0):
            case '-i': ignorecase = True
            case '-h': usage(0)     # Discuss: exit codes
            case _   : usage(1)

    # Process standard input
    for line in stream:
        if ignorecase:
            line = line.lower()

        # Discuss: splitting string
        word1, word2 = line.split()
        if is_anagram(word1, word2):
            print('ANAGRAM!')
        else:
            print('NOT ANAGRAM!')

Python: Leetspeak¶

def leetspeak(text: str, mapping: dict[str, str]) -> str:
  ''' Translate each line of text into leetspeak '''
  result = []                           # TODO: Add doctest???
  for letter in text:                   # Discuss: get
      result.append(mapping.get(letter.lower(), letter))
  return ''.join(result)                # Discuss: join

# Main Execution

def main(arguments: list[str]=sys.argv[1:], stream=sys.stdin):
    fr_chars = 'aeio'
    to_chars = '4310'

    while arguments and arguments[0].startswith('-'):
        match argument := arguments.pop(0):
            case '-f': fr_chars = arguments.pop(0)
            case '-t': to_chars = arguments.pop(0)
            case '-h': usage(0)
            case _   : usage(1)

    mapping = {}                        # Discuss: zip
    for fr, to in zip(fr_chars, to_chars):
        mapping[fr] = to

    for line in stream:
        print(leetspeak(line, mapping), end='')

    # Variant: Stream output to cowsay process
    with subprocess.Popen(['cowsay'], stdin=subprocess.PIPE, text=True) as process:
        for line in stream:
            process.stdin.write(leetspeak(line, mapping))
