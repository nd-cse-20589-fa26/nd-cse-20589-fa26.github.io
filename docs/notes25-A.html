Echo Server¶

#!/usr/bin/env python3

import socket
import sys

# Constants

BUFSIZ = (1<<10)

# Parse command line arguments

try:
    host = '127.0.0.1'
    port = int(sys.argv[1])
except IndexError:
    print('Usage: echo_server.py port', file=sys.stderr)
    sys.exit(1)

# Connect to remote machine at host and port

with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server:
    # Bind to port and begin listening
    server.bind((host, port))
    server.listen()
    print(f'Listening on {host}:{port}')

    while True:
        # Accept a client
        client, address = server.accept()

        with client:
            print(f'Accepted connection from {address[0]}:{address[1]}')

            # Read from client and send it back
            while request := client.recv(BUFSIZ):
                print(request.decode(), end='')
                client.send(request)

  4. Host indicates the server we want to contact.

  5. Port specifies the service number we want to utilize.

#!/usr/bin/env python3

import socket
import ssl
import sys

from urllib.parse import urlsplit

# Parse command line arguments

try:
    url = urlsplit(sys.argv[1])
except IndexError:
    print('Usage: http_client.py URL', file=sys.stderr)
    sys.exit(1)

# Create socket (wrap with ssl if scheme is HTTPS)

socket = socket.create_connection((url.hostname, url.port))
if url.scheme == 'https':
    context = ssl.create_default_context()
    socket  = context.wrap_socket(socket, server_hostname=url.hostname)

with socket:
    # Send HTTP Request
    with socket.makefile('w', encoding='utf-8') as wfile:
        wfile.write(f'GET {url.path} HTTP/1.0\r\n')
        wfile.write(f'Host: {url.hostname}\r\n')
        wfile.write('\r\n')

    # Read HTTP Response
    with socket.makefile('r', encoding='utf-8') as rfile:
        while line := rfile.readline().strip():
            print(line)

        print()
        print(rfile.read())
