Python: Repeated Execution¶

# 7. Repeated Execution

$ $EDITOR roll.py                         # Discuss: while loop

while n != 5:                             # Discuss: infinite loop
    print(n)
    n = random.randint(0, 10)

...

n = None                                  # Discuss: None
while n != 5:
    n = random.randint(0, 10)
    print(n)

...

while (n := random.randint(0, 10)) != 5:  # Discuss: walrus
    print(n)

Python: Fizz Buzz (Base)¶

$ $EDITOR fizzbuzz.py
#!/usr/bin/env python3

for number in range(1, 101):            # Discuss: Fizz Buzz
    if number % 3 == 0 and number % 5 == 0:
        print('FizzBuzz')
    elif number % 3 == 0:
        print('Fizz')
    elif number % 5 == 0:
        print('Buzz')
    else:
        print(number)

Python: Fizz Buzz (Functions)¶

$ $EDITOR fizzbuzz.py
#!/usr/bin/env python3

def fizzbuzz(start=1, stop=100):        # Discuss: keyword arguments
    for number in range(start, stop + 1):
        if number % 3 == 0 and number % 5 == 0:
            print('FizzBuzz')
        elif number % 3 == 0:
            print('Fizz')
        elif number % 5 == 0:
            print('Buzz')
        else:
            print(number)

fizzbuzz(1, 20)
fizzbuzz()                              # Discuss: default arguments
fizzbuzz(end=5)
fizzbuzz(end=5, start=2)
